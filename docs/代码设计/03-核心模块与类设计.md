## 一、存储层：SessionStorage / LaneManager

对应 [02-树形对话历史系统](../系统设计/02-数据与存储层/02-树形对话历史系统.md)、[03-Lane分支管理系统](../系统设计/02-数据与存储层/03-Lane分支管理系统.md)、[12-存储层设计](../系统设计/02-数据与存储层/12-存储层设计.md)。

```python
# src/storage/session_storage.py
import asyncio
from pathlib import Path
from typing import Optional
from .models import Entry

class SessionStorage:
    """单个 session 的树形历史：内存索引 + JSONL 落盘"""

    def __init__(self, session_id: str, data_dir: Path = Path("data/sessions")):
        self.session_id = session_id
        self.path = data_dir / f"{session_id}.jsonl"
        self._entries: dict[str, Entry] = {}
        self._children_index: dict[str, list[str]] = {}
        self._next_seq = 1
        self._write_lock = asyncio.Lock()   # 单 session 单写锁，见 03 号文档 7.3 节
        self._load()

    def _load(self) -> None:
        """启动时全量加载 + 建索引，跳过半截损坏行（见 12 号文档 6.1 节）"""
        ...

    async def append_message(self, entry: Entry) -> Entry:
        """分配 seq、写 JSONL、更新内存索引；持写锁"""
        async with self._write_lock:
            entry.seq = self._next_seq
            self._next_seq += 1
            self._append_to_file(entry)
            self._entries[entry.id] = entry
            if entry.parent is not None:
                self._children_index.setdefault(entry.parent, []).append(entry.id)
            return entry

    def get_entry(self, entry_id: str) -> Optional[Entry]:
        return self._entries.get(entry_id)

    def get_children(self, entry_id: str) -> list[str]:
        return self._children_index.get(entry_id, [])

    def get_history_path(self, leaf_id: str) -> list[Entry]:
        """从叶子沿 parent 走到根，返回根→叶顺序；只用内存索引，不扫文件"""
        ...

    def find_common_ancestor(self, id_a: str, id_b: str) -> Optional[str]:
        """对应 02 号文档 3.3 节算法：分别取两条到根的路径，找最后一个共同节点"""
        ...

    def get_context_window(self, leaf_id: str, max_tokens: int) -> list[Entry]:
        """对应 04 号文档 5.1 节：从叶子向上加载，直到 token 预算耗尽"""
        ...
```

```python
# src/storage/lane_manager.py
from pathlib import Path
from .models import LanePointer

class LaneManager:
    """Lane 指针管理：append-only 流水 + 重放取最新（见 12 号文档 5.3/6.2 节）"""

    def __init__(self, session_id: str, data_dir: Path = Path("data/sessions")):
        self.session_id = session_id
        self.path = data_dir / f"{session_id}_lanes.jsonl"
        self._lanes: dict[str, LanePointer] = {}
        self._load()   # 重放全部行，同名 lane 以最后一次为准

    def create_lane(self, name: str, from_id: str, description: str = "") -> LanePointer:
        if name in self._lanes:
            raise ValueError(f"分支已存在: {name}")
        pointer = LanePointer(lane=name, leaf_id=from_id, seq=1, created_from=from_id, description=description)
        self._append(pointer)
        return pointer

    def update_lane(self, name: str, leaf_id: str) -> LanePointer:
        prev = self._lanes.get(name)
        seq = (prev.seq + 1) if prev else 1
        pointer = LanePointer(lane=name, leaf_id=leaf_id, seq=seq,
                               created_from=prev.created_from if prev else None,
                               description=prev.description if prev else "")
        self._append(pointer)
        return pointer

    def get_lane(self, name: str) -> LanePointer:
        if name not in self._lanes:
            raise KeyError(f"分支不存在: {name}")
        return self._lanes[name]

    def list_lanes(self) -> list[LanePointer]:
        return list(self._lanes.values())

    def delete_lane(self, name: str, current_lane: str) -> None:
        if name == "main":
            raise ValueError("不能删除 main 分支")
        if name == current_lane:
            raise ValueError("不能删除当前正在使用的分支")
        del self._lanes[name]   # 只删指针记录，不影响 SessionStorage 里的节点

    def _append(self, pointer: LanePointer) -> None:
        self._lanes[pointer.lane] = pointer   # 内存态直接覆盖
        with open(self.path, "a", encoding="utf-8") as f:
            f.write(pointer_to_json(pointer) + "\n")   # 物理文件永远 append，不覆盖
```

**`compare_lanes` 放在哪一层**：不放进 `LaneManager`,因为它需要同时用到 `SessionStorage.find_common_ancestor` 和 `LaneManager.get_lane`,属于两个存储对象的协作查询。放在 `Agent`（或 API 路由层的一个独立函数 `compare_lanes(storage, lane_manager, lane_a, lane_b)`）里组合调用,`LaneManager` 和 `SessionStorage` 各自保持职责单一,不互相 import。

## 二、MessageProvider 抽象与 Agent 主循环

对应 [04-Agent主循环](../系统设计/03-Agent核心层/04-Agent主循环.md) 4.0/4.1/4.2 节。这是全项目唯一要求"第一版代码就落地"的接口,写法直接照抄功能设计文档的示例,不做改动：

```python
# src/agent/message_provider.py
from typing import Protocol
from ..llm.events import Message
from ..storage.session_storage import SessionStorage
from ..storage.lane_manager import LaneManager

class MessageProvider(Protocol):
    def get_context(self) -> list[Message]: ...
    def append(self, message: Message) -> None: ...

class TreeMessageProvider:
    def __init__(self, storage: SessionStorage, lane_manager: LaneManager, lane: str):
        self.storage = storage
        self.lane_manager = lane_manager
        self.lane = lane

    def get_context(self) -> list[Message]:
        leaf_id = self.lane_manager.get_lane(self.lane).leaf_id
        entries = self.storage.get_context_window(leaf_id, max_tokens=8000)
        return [m for e in entries for m in _entry_to_messages(e)]   # 一对多展开，见 01 号文档八节

    def append(self, message: Message) -> None:
        leaf_id = self.lane_manager.get_lane(self.lane).leaf_id
        entry = self.storage.append_message(_message_to_entry(message, parent=leaf_id, lane=self.lane))
        self.lane_manager.update_lane(self.lane, entry.id)

class EphemeralMessageProvider:
    def __init__(self, seed_task: str):
        self._messages: list[Message] = [Message(role="user", content=seed_task)]

    def get_context(self) -> list[Message]:
        return self._messages

    def append(self, message: Message) -> None:
        self._messages.append(message)
```

```python
# src/tools/base.py
from abc import ABC, abstractmethod
from typing import Any
from ..permission.manager import PermissionLevel
from .result import ToolResult   # 见 01 号文档四节

class Tool(ABC):
    name: str
    description: str
    parameters: dict[str, Any]     # JSON Schema
    permission_level: PermissionLevel

    @abstractmethod
    async def execute(self, **kwargs) -> ToolResult: ...

    def get_schema(self) -> dict:
        return {"type": "function", "function": {
            "name": self.name, "description": self.description, "parameters": self.parameters
```

`ToolRegistry` 是工具的统一注册与调用入口，`execute` 用 `async def`，参数校验失败时抛 `ValidationError` 而不是裸异常：

```python
# src/tools/registry.py
class ToolRegistry:
    def __init__(self):
        self._tools: dict[str, Tool] = {}

    def register(self, tool: Tool) -> None: ...
    def get_tool(self, name: str) -> Tool: ...
    def get_tool_schemas(self) -> list[dict]: ...

    async def execute(self, name: str, args: dict) -> ToolResult:
        tool = self._tools.get(name)
        if tool is None:
            raise ToolExecutionError(message=f"工具不存在: {name}", code="TOOL_NOT_FOUND", tool_name=name, tool_args=args)
        errors = validate_args(tool.parameters, args)
        if errors:
            raise ValidationError(message="参数校验失败", code="VALIDATION_ERROR", tool_name=name, validation_errors=errors)
        return await tool.execute(**args)

    @staticmethod
    def default(workspace: Path) -> "ToolRegistry":
        """构造包含 6 个核心工具的默认注册表"""
        ...

    @staticmethod
    def readonly(workspace: Path) -> "ToolRegistry":
        """构造只读子集（read_file/glob/grep），供子 Agent 使用（15 号文档第四节）"""
        ...
```

## 五、权限系统：PermissionManager

对应 [09-权限控制系统](../系统设计/04-支撑模块/09-权限控制系统.md)、[10-沙箱隔离](../系统设计/04-支撑模块/10-沙箱隔离.md)。

```python
# src/permission/manager.py
from typing import Callable, Optional, Awaitable
from .rules import is_safe_path, is_dangerous_command
from .models import PermissionDecision, PermissionLevel

TOOL_PERMISSIONS: dict[str, PermissionLevel] = {
    "read_file": PermissionLevel.SAFE, "glob": PermissionLevel.SAFE, "grep": PermissionLevel.SAFE,
    "write_file": PermissionLevel.WRITE, "edit_file": PermissionLevel.WRITE,
    "bash": PermissionLevel.DANGEROUS,
    "delegate_task": PermissionLevel.SAFE,   # 只读子 Agent，天然 SAFE（15 号文档五节）
}

class PermissionManager:
    def __init__(self, workspace: str, config: dict | None = None):
        self.workspace = workspace
        self.config = config or {}
        self.auto_approved: set[str] = set()
        self.denied: set[str] = set()
        self.ask_user_callback: Optional[Callable[[dict], Awaitable[dict]]] = None

    async def check(self, tool_name: str, args: dict) -> PermissionDecision:
        level = TOOL_PERMISSIONS.get(tool_name, PermissionLevel.DANGEROUS)
        fingerprint = self._fingerprint(tool_name, args)
        if fingerprint in self.denied:
            return PermissionDecision(allowed=False, reason="用户之前拒绝了此操作")
        if fingerprint in self.auto_approved:
            return PermissionDecision(allowed=True, reason="自动批准（用户之前允许）", auto_approved=True)
        if level is PermissionLevel.SAFE:
            return PermissionDecision(allowed=True, reason="只读操作，自动允许", auto_approved=True)
        if level is PermissionLevel.WRITE:
            return await self._check_write(tool_name, args)
        return await self._check_dangerous(tool_name, args, fingerprint)

    async def _check_write(self, tool_name: str, args: dict) -> PermissionDecision: ...
    async def _check_dangerous(self, tool_name: str, args: dict, fingerprint: str) -> PermissionDecision: ...
    async def _ask_user(self, **kwargs) -> PermissionDecision: ...
    def _fingerprint(self, tool_name: str, args: dict) -> str: ...
```

`is_safe_path`/`is_dangerous_command`/黑名单路径列表全部放进 `src/permission/rules.py`,和 `PermissionManager` 分开,方便单独做单元测试(对齐 [09 号文档](../系统设计/04-支撑模块/09-权限控制系统.md) 五、六节的独立函数)。

`BashTool.execute` 内部调用的 `run_command()` 直接复用 [10-沙箱隔离](../系统设计/04-支撑模块/10-沙箱隔离.md) 第五节给出的实现,放进 `src/tools/exec_tool.py`,权限检查在 `Agent._execute_tool_calls` 里前置调用,不在工具内部重复检查(避免"谁调用就检查一次"的重复逻辑,呼应 [15 号文档](../系统设计/03-Agent核心层/15-子Agent系统.md) 第八节"权限检查发生在工具执行入口"的原则)。

## 六、子 Agent：DelegateTaskTool

对应 [15-子Agent系统](../系统设计/03-Agent核心层/15-子Agent系统.md)。

```python
# src/tools/subagent_tool.py
from .base import Tool
from .result import ToolResult
from ..permission.manager import PermissionLevel
from ..agent.agent import Agent
from ..agent.message_provider import EphemeralMessageProvider

MAX_DEPTH = 1
SUMMARY_LIMIT = 2000

class DelegateTaskTool(Tool):
    name = "delegate_task"
    description = "把需要多步调查才能有结论的子任务委托给独立子 Agent，只回结论摘要"
    parameters = {
        "type": "object",
        "properties": {
            "task": {"type": "string", "description": "交给子 Agent 的任务描述"},
            "max_steps": {"type": "integer", "description": "最多执行的工具调用轮数，默认 8"},
        },
        "required": ["task"],
    }
    permission_level = PermissionLevel.SAFE

    def __init__(self, session_id: str, config: dict, depth: int):
        self.session_id = session_id
        self.config = config
        self.depth = depth

    async def execute(self, task: str, max_steps: int = 8) -> ToolResult:
        if self.depth >= MAX_DEPTH:
            return ToolResult(is_error=True, content="子 Agent 不允许递归派生")

        sub_agent = Agent(
            session_id=self.session_id,
            config=self.config,
            provider=EphemeralMessageProvider(seed_task=task),
            depth=self.depth + 1,
        )
        sub_agent.tool_registry = ToolRegistry.readonly(self.config["workspace"])

        try:
            result = await asyncio.wait_for(sub_agent.run(max_steps=max_steps), timeout=120)
        except asyncio.TimeoutError:
            return ToolResult(is_error=True, content="[error] 子 Agent 执行超时")

        return self._package_result(result)

    def _package_result(self, result) -> ToolResult:
        summary = result.final_text[:SUMMARY_LIMIT]
        return ToolResult(
            content=f"[{result.status}] {summary}",
            metadata={"tool_calls": result.step_count, "files_touched": result.touched_paths},
        )
```

**并行派发多个子任务**（[15 号文档](../系统设计/03-Agent核心层/15-子Agent系统.md) 7.2 节,P2 加分项）放在 `Agent._execute_tool_calls` 里判断:如果一次 assistant 回复里有多个 `delegate_task` 调用,用 `asyncio.Semaphore(3)` 包一层,每个协程内部各自 `try/except` 兜底,不依赖 `gather(return_exceptions=True)`——这个坑已经在功能设计文档里点明,代码实现时直接照抄该节的写法,不再重复设计。

```python
# src/observability/logger.py
from pathlib import Path
from enum import Enum
from typing import Callable, Optional, Any

class LogLevel(str, Enum):
    DEBUG = "DEBUG"; INFO = "INFO"; WARNING = "WARNING"; ERROR = "ERROR"

class StructuredLogger:
    def __init__(self, session_id: str, log_dir: Path = Path("logs")):
        self.session_id = session_id
        self.agent_log = log_dir / f"{session_id}_agent.jsonl"
        self.ui_callback: Optional[Callable[[dict], None]] = None   # 由 api/ws.py 注入

    def log(self, level: LogLevel, event: str, **fields: Any) -> None:
        entry = {"timestamp": _now_iso(), "session_id": self.session_id, "level": level.value, "event": event, **fields}
        self._write_to_file(entry)
        if self.ui_callback:
            self.ui_callback(entry)

    def info(self, event: str, **fields): self.log(LogLevel.INFO, event, **fields)
    def warning(self, event: str, **fields): self.log(LogLevel.WARNING, event, **fields)
    def error(self, event: str, **fields): self.log(LogLevel.ERROR, event, **fields)
    def debug(self, event: str, **fields): self.log(LogLevel.DEBUG, event, **fields)
```

事件名固定使用 [11 号文档](../系统设计/04-支撑模块/11-日志与可观测性.md) 4.1 节表格里的名字（`run_started`/`context_loaded`/`llm_request`/`llm_response`/`run_completed`/`run_error`）,`ui_callback` 由 `src/api/ws.py` 在建立连接时注入,把日志事件转成 WebSocket 的 `status_update`/相应事件推送出去,不需要单独一套"日志转发管道"。**不实现**日志轮转（`LogRotator`）和 Web UI 独立日志查看器——对应 [14-技术决策与取舍](../系统设计/06-项目管理/14-技术决策与取舍.md) 4.4 节和 [07-Web-UI设计](../系统设计/05-Web-UI层/07-Web-UI设计.md) 第九节已经明确砍掉的范围,这里不重复实现。

```python
# src/api/ws.py（核心骨架）
from fastapi import APIRouter, WebSocket

router = APIRouter()

@router.websocket("/ws/{session_id}")
async def websocket_endpoint(ws: WebSocket, session_id: str):
    await ws.accept()
    agent = _get_or_create_agent(session_id)
    agent.logger.ui_callback = lambda entry: _forward_log_as_event(ws, entry)
    agent.permission_manager.ask_user_callback = lambda req: _ask_user_via_ws(ws, req)

    async for raw in ws.iter_json():
        if raw["type"] == "send_message":
            await _handle_send_message(ws, agent, raw)
        elif raw["type"] == "permission_response":
            _resolve_pending_permission(raw)
```

`_ask_user_via_ws` 的实现要点：发出 `permission_request` 事件后,挂起一个 `asyncio.Future`,等对应 `request_id` 的 `permission_response` 到达时 `set_result`——这是权限确认"异步等待用户点击"这件事在代码里唯一需要小心处理并发的地方,`request_id` 必须能在多个并发权限请求间正确路由。
